{{- /* A click-to-load video: nothing is fetched from YouTube or Vimeo until the visitor presses play. The still is the video's own thumbnail, fetched once and kept in assets/media/video/ (yt-.jpg, vimeo-.jpg); without one the frame stays black with the title. */ -}} {{- $title := .title | default "Video" -}} {{- $still := "" -}} {{- with findRESubmatch `youtube(?:-nocookie)?\.com/embed/([A-Za-z0-9_-]+)` .src 1 }}{{ $still = printf "media/video/yt-%s.jpg" (index (index . 0) 1) }}{{ end -}} {{- with findRESubmatch `player\.vimeo\.com/video/([0-9]+)` .src 1 }}{{ $still = printf "media/video/vimeo-%s.jpg" (index (index . 0) 1) }}{{ end -}} {{- $img := "" }}{{ with $still }}{{ $img = resources.Get . }}{{ end -}}
{{ with $img }}{{ $s := .Resize "800x webp q80" }}{{ $l := .Resize "1280x webp q80" }}{{ end }} {{ partial "icon.html" "play" }} {{ if not $img }}{{ $title }}{{ end }}
